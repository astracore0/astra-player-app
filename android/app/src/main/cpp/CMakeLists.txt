cmake_minimum_required(VERSION 3.18.1)
project(astra_whisper C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/whisper.cpp ${CMAKE_BINARY_DIR}/whisper_build)
add_library(astra_whisper SHARED whisper_jni.cpp)
target_include_directories(astra_whisper PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/whisper.cpp/include ${CMAKE_CURRENT_SOURCE_DIR}/whisper.cpp/ggml/include)
target_link_libraries(astra_whisper whisper log)
